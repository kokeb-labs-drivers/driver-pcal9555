31[[nodiscard]] constexpr uint16_t Apply(uint16_t value, uint16_t mask, bool set) noexcept {
32 const auto v = static_cast<uint32_t>(value);
33 const auto m = static_cast<uint32_t>(mask);
34 return static_cast<uint16_t>(set ? (v | m) : (v & ~m));
35}

39[[nodiscard]] constexpr uint32_t ApplyDrive(uint32_t codes, uint16_t mask,
40 DriveStrength level) noexcept {
41 const uint32_t code = static_cast<uint32_t>(ToUnderlying(level)) & reg::kDriveCodeMask;
42 for (uint32_t pin = 0U; pin < kPinCount; ++pin) {
43 if ((static_cast<uint32_t>(mask) & (uint32_t{1} << pin)) != 0U) {
44 const uint32_t shift = reg::kDriveBitsPerPin * pin;
45 codes = (codes & ~(reg::kDriveCodeMask << shift)) | (code << shift);
46 }
47 }
48 return codes;
49}

58[[nodiscard]] constexpr uint16_t WithByte(uint16_t value, uint32_t port, uint8_t byte) noexcept {

117[[nodiscard]] constexpr uint32_t UncertainIndex(uint8_t command) noexcept {
118 const auto c = static_cast<uint32_t>(command);
119 if (c == static_cast<uint32_t>(reg::kOutputPortConfig)) {
120 return 10U;
121 }
122 if (c >= static_cast<uint32_t>(reg::kDriveStrength0)) {
123 return (4U + ((c - static_cast<uint32_t>(reg::kDriveStrength0)) / 2U)) & 0x0FU;
124 }
125 return (c / 2U) & 0x0FU;
126}

137[[nodiscard]] constexpr uint32_t StrongerDrive(uint32_t a, uint32_t b) noexcept {
138 uint32_t codes = 0U;
139 for (uint32_t pin = 0U; pin < kPinCount; ++pin) {
140 const uint32_t shift = reg::kDriveBitsPerPin * pin;
141 const uint32_t ca = (a >> shift) & reg::kDriveCodeMask;
142 const uint32_t cb = (b >> shift) & reg::kDriveCodeMask;
143 codes |= ((ca > cb) ? ca : cb) << shift;
144 }
145 return codes;
146}

170[[nodiscard]] constexpr uint16_t RegisterValue(const Config& c, uint8_t command) noexcept {
171 uint32_t v = 0U;
172 if (command == reg::kOutputPort0) {
173 v = c.output_high_mask;
174 } else if (command == reg::kPolarityPort0) {
175 v = c.polarity_invert_mask;
176 } else if (command == reg::kConfigPort0) {
177 v = c.input_mask;
178 } else if (command == reg::kDriveStrength0) {
179 v = c.drive_strength_codes & 0xFFFFU;
180 } else if (command == reg::kDriveStrength2) {
181 v = c.drive_strength_codes >> 16U;
182 } else if (command == reg::kInputLatch0) {
183 v = c.input_latch_mask;
184 } else if (command == reg::kPullEnable0) {
185 v = c.pull_enable_mask;
186 } else if (command == reg::kPullSelect0) {
187 v = c.pull_up_mask;
188 } else if (command == reg::kInterruptMask0) {
189 v = c.interrupt_mask;
190 } else if (command == reg::kOutputPortConfig) {
191 v = OutputPortConfig(c);
192 } else {
193 // Not a register the driver writes.
194 }
195 return static_cast<uint16_t>(v);
196}

199constexpr void Adopt(Config& c, uint8_t command, uint16_t value) noexcept {
200 const auto v = static_cast<uint32_t>(value);
201 if (command == reg::kOutputPort0) {
202 c.output_high_mask = value;
203 } else if (command == reg::kPolarityPort0) {
204 c.polarity_invert_mask = value;
205 } else if (command == reg::kConfigPort0) {
206 c.input_mask = value;
207 } else if (command == reg::kDriveStrength0) {
208 c.drive_strength_codes = (c.drive_strength_codes & 0xFFFF'0000U) | v;
209 } else if (command == reg::kDriveStrength2) {
210 c.drive_strength_codes = (c.drive_strength_codes & 0x0000'FFFFU) | (v << 16U);
211 } else if (command == reg::kInputLatch0) {
212 c.input_latch_mask = value;
213 } else if (command == reg::kPullEnable0) {
214 c.pull_enable_mask = value;
215 } else if (command == reg::kPullSelect0) {
216 c.pull_up_mask = value;
217 } else if (command == reg::kInterruptMask0) {
218 c.interrupt_mask = value;
219 } else if (command == reg::kOutputPortConfig) {
220 c.port0_mode = ((v & static_cast<uint32_t>(reg::kOutputConfigOden0)) != 0U)
221 ? OutputMode::OpenDrain
222 : OutputMode::PushPull;
223 c.port1_mode = ((v & static_cast<uint32_t>(reg::kOutputConfigOden1)) != 0U)
224 ? OutputMode::OpenDrain
225 : OutputMode::PushPull;
226 } else {
227 // Not a register the driver writes: nothing to adopt.
228 }

249[[nodiscard]] constexpr ResetSentinel FindResetSentinel(const Config& c, bool agile,
250 uint32_t uncertain = 0U) noexcept {
251 const auto drive_low = static_cast<uint16_t>(c.drive_strength_codes & 0xFFFFU);
252 const auto drive_high = static_cast<uint16_t>(c.drive_strength_codes >> 16U);
253 const auto drive_reset = static_cast<uint16_t>(reg::kDriveStrengthReset & 0xFFFFU);
254 const std::array<ResetSentinel, 10> candidates{{
255 Candidate(reg::kInterruptMask0, c.interrupt_mask, reg::kInterruptMaskReset),
256 Candidate(reg::kPullEnable0, c.pull_enable_mask, reg::kPullEnableReset),
257 Candidate(reg::kPullSelect0, c.pull_up_mask, reg::kPullSelectReset),
258 Candidate(reg::kInputLatch0, c.input_latch_mask, reg::kInputLatchReset),
259 Candidate(reg::kDriveStrength0, drive_low, drive_reset),
260 Candidate(reg::kDriveStrength2, drive_high, drive_reset),
261 Candidate(reg::kPolarityPort0, c.polarity_invert_mask, reg::kPolarityReset),
262 Candidate(reg::kOutputPortConfig, OutputPortConfig(c), reg::kOutputPortConfigReset, false),
263 Candidate(reg::kConfigPort0, c.input_mask, reg::kConfigReset),
264 Candidate(reg::kOutputPort0, c.output_high_mask, reg::kOutputReset),
265 }};
266 ResetSentinel found{.reg = kNoSentinel, .value = 0U, .reset = 0U, .pair = true, .only = true};
267 for (const ResetSentinel& candidate : candidates) {
268 if (!reg::IsImplemented(candidate.reg, agile) || (candidate.value == candidate.reset) ||
269 ((uncertain & UncertainBit(candidate.reg)) != 0U)) {
270 continue; // an unknown register proves nothing either way
271 }
272 if (found.reg != kNoSentinel) {
273 found.only = false;
274 break;
275 }
276 found = candidate;
277 }
278 if (uncertain != 0U) {
279 found.only = false; // an unknown register may differ too
280 }
281 return found;
282}
