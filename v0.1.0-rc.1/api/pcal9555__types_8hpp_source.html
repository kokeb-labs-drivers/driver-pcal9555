283constexpr uint32_t DriveStrengthAll(DriveStrength level) noexcept {
284 // Two bits per code: a value that is not an enumerator is masked, never shifted into the next
285 // pin.
286 const uint32_t code = static_cast<uint32_t>(level) & 0x3U;
287 uint32_t codes = 0U;
288 for (uint32_t pin = 0U; pin < kPinCount; ++pin) {
289 codes |= code << (2U * pin);
290 }
291 return codes;
292}

337constexpr Error Validate(const Config& config) noexcept {
338 if ((static_cast<uint32_t>(config.variant) > static_cast<uint32_t>(Variant::Pcal9555a)) ||
339 (static_cast<uint32_t>(config.port0_mode) > static_cast<uint32_t>(OutputMode::OpenDrain)) ||
340 (static_cast<uint32_t>(config.port1_mode) > static_cast<uint32_t>(OutputMode::OpenDrain))) {
341 return Error::InvalidArgument;
342 }
343 if (config.variant == Variant::Pca9555) {
344 // Every field but the four both parts have must be at its power-up value. The comparison is
345 // of the whole struct, so a field added later is covered without a change here.
346 Config agile = config;
347 const Config reset{};
348 agile.output_high_mask = reset.output_high_mask;
349 agile.polarity_invert_mask = reset.polarity_invert_mask;
350 agile.input_mask = reset.input_mask;
351 if (!(agile == reset)) {
352 return Error::Unsupported;
353 }
354 }
355 return Error::Ok;
356}
